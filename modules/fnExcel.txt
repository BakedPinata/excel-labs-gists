/*
Description:  Checks if supplied value(s) have no length (empty)
Notes:        values_to_check can be single value, row, column or 2D array of values
              Optional check_all to require all values null (no effect if single value)
Sources:      @BakedPinata
*/
ISNULL = LAMBDA(values_to_check, [check_all],
  LET(
    is_empty_array, MAP(values_to_check, LAMBDA(check_value, LEN(check_value) = 0)),
    final_check_all, N(check_all)=1,
    IF(final_check_all, AND(is_empty_array), OR(is_empty_array))
  )
);

/*
Description:  Returns supplied value if not empty
Notes:        Option to specify alternate value, otherwise defaults to empty string
              Useful for wrapping XLOOKUPs that return 0 instead of "" on returning an empty cell
Sources:      @BakedPinata
*/
IFNULL = LAMBDA(value_to_check, [value_if_null],
  IF(LEN(value_to_check) > 0,
    value_to_check,
    IF(ISOMITTED(value_if_null), "", value_if_null)
  )
);

// Return a date value from a string value in the format 'dd/MM/yyy'
DATEFROMSTRING = LAMBDA(date_string, DATE(RIGHT(date_string, 4), MID(date_string, 4, 2), LEFT(date_string, 2)));

// Source: https://www.reddit.com/r/excel/comments/17ovi3x/comment/k85dtks/?utm_source=share&utm_medium=web3x&utm_name=web3xcss&utm_term=1&utm_content=share_button
CALENDAR = LAMBDA(Year, Month, Day, [StartDay],
    LET(
        SD, IF(ISOMITTED(StartDay), 2, StartDay),
        INPUT, DATE(Year, Month, Day),
        WEEKDAY_OFFSET, WEEKDAY(INPUT, SD),
        DAY_NAMES, EXPAND(TEXT(SEQUENCE(7,,SD,1), "dddd"), 6 + WEEKDAY_OFFSET, , ""),
        DAY_NUMBERS, DAY(SEQUENCE(EOMONTH(INPUT, 0) - INPUT + 1, , INPUT)),
        MONTH_NAME, EXPAND(UPPER(TEXT(INPUT, "MMMM")), 7, , ""),
        OUTPUT, WRAPROWS(VSTACK(MONTH_NAME, DAY_NAMES, DAY_NUMBERS), 7, ""),
        OUTPUT
    )
);

// See https://learn.microsoft.com/en-us/troubleshoot/microsoft-365-apps/excel/determine-a-leap-year
// Accepts date/datetime, date/datetime serial, year integer, date string and year string input formats
ISLEAPYEAR = LAMBDA(input_value,
  IF(
    LEN(input_value) = 0,
    "",
    LET(
      number_value, input_value * 1,
      year_value, YEAR(IFS(number_value < 1, NA(), number_value > 4000, number_value, number_value <= 4000, DATE(number_value, 1, 1))),
      OR(MOD(year_value, 400) = 0, AND(MOD(year_value, 4) = 0, MOD(year_value, 100) <> 0))
    )
  )
)

// Source: https://www.reddit.com/r/excel/comments/17ovi3x/comment/k85c1s9/?utm_source=share&utm_medium=web3x&utm_name=web3xcss&utm_term=1&utm_content=share_button
SUBTITUTEALL = LAMBDA(Text_to_Change, Substitution_Table,
    LET(
        A, " " & Text_to_Change & " ",
        B, TRIM(Substitution_Table),
        Prefix, {"-", """", "'", " "},
        Suffix, {"-", """", "'", " ", ".", ",", ":", ";", "=", "?", "!"},
        Frm_1, TOCOL(Prefix & TOCOL(CHOOSECOLS(B, 1) & Suffix)),
        Frm_2, VSTACK(UPPER(Frm_1), LOWER(Frm_1), PROPER(Frm_1)),
        To_1, TOCOL(Prefix & TOCOL(CHOOSECOLS(B, 2) & Suffix)),
        To_2, VSTACK(UPPER(To_1), LOWER(To_1), PROPER(To_1)),
        Output, REDUCE(A, SEQUENCE(ROWS(To_2)), LAMBDA(X, Y, SUBSTITUTE(X, INDEX(Frm_2, Y), INDEX(To_2, Y)))),
        TRIM(Output)
    )
);

// Truncate text at the last specified delimiter before a specified max length
TRUNCATEATDELIMITER = LAMBDA(text, maxLength, delimiter,
    IF(
        LEN(text) <= maxLength,
        text,
        LET(
            trimmedText, LEFT(text, maxLength + 1),
            lastDelimiterPosition, FIND(
                "@||@",
                SUBSTITUTE(
                    trimmedText,
                    delimiter,
                    "@||@",
                    LEN(trimmedText) - LEN(SUBSTITUTE(trimmedText, delimiter, ""))
                )
            ),
            IF(ISERROR(lastDelimiterPosition), LEFT(trimmedText, maxLength), LEFT(trimmedText, lastDelimiterPosition - 1))
        )
    )
);
